落到代码：令牌映射 // gui_theme_t

typedef struct gui_theme_t {
    /* 分层：无阴影，靠 surface 阶梯 + 1px 线 */
    gui_color_t bg, surface, surface2, surface3;
    gui_color_t line, line2;              /* 发丝 / 强描边 */
    /* 文字层级 */
    gui_color_t fg, fg2, dim;
    /* 单色荧光强调 */
    gui_color_t accent, accent_fg, accent_hover;
    /* 语义 */
    gui_color_t ok, warn, err, sel;
    /* 几何：形状令牌化，禁止各应用自定义 */
    int          radius, radius_small;        /* 0..2 */
    float        shadow_alpha;               /* = 0.0 : 本方案禁用阴影 */
} gui_theme_t;

/* 主题表：dark = 磷光终端，light = 高对比纸面 */
static const gui_theme_t GUI_THEME_DARK = {
    .bg = 0xFF0B0F10, .surface = 0xFF0F1517, .accent = 0xFF00E0A4, /* 0xAARRGGBB */
    .radius = 2, .radius_small = 1, .shadow_alpha = 0.0f,
};
应用只引用令牌，不写字面量颜色

gui_rect(x, y, w, h, T->surface);   gui_frame(x, y, w, h, T->line, 1);   gui_text(..., T->fg);   gui_text(..., T->dim); —— 13 个应用共享同一份 gui_theme_t， 换主题 = 换一张表，零改应用代码。
